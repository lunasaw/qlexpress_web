## 3. 表达式渲染策略 (核心复用点)

表达式是整个系统中复用度最高的组件，采用递归渲染模式。

### 3.1 表达式渲染器

```typescript
// ExpressionRenderer.tsx
const ExpressionRenderer: React.FC<ExpressionComponentProps> = ({
  expression,
  depth = 0,
  compact = false,
  highlightVariables = [],
  onVariableClick
}) => {
  // 根据表达式类型分发到对应子渲染器
  switch (expression.type) {
    case 'literal':
      return <LiteralRenderer value={expression.value} />;

    case 'variable':
      return (
        <VariableRenderer
          name={expression.name}
          highlighted={highlightVariables.includes(expression.name)}
          onClick={() => onVariableClick?.(expression.name)}
        />
      );

    case 'operator':
      return (
        <OperatorRenderer
          operator={expression.operator}
          left={expression.left}
          right={expression.right}
          depth={depth}
          compact={compact}
        />
      );

    case 'method':
      return (
        <MethodCallRenderer
          target={expression.target}
          methodName={expression.methodName}
          arguments={expression.arguments}
          depth={depth}
        />
      );

    case 'static_method':
      return (
        <StaticMethodRenderer
          className={expression.className}
          methodName={expression.methodName}
          arguments={expression.arguments}
        />
      );

    case 'new':
      return (
        <NewExpressionRenderer
          className={expression.className}
          arguments={expression.arguments}
        />
      );

    case 'cast':
      return (
        <CastRenderer
          targetType={expression.targetType}
          expression={expression.expression}
          depth={depth}
        />
      );

    case 'field_access':
      return (
        <FieldAccessRenderer
          object={expression.object}
          field={expression.field}
          depth={depth}
        />
      );

    case 'ternary':
      return (
        <TernaryRenderer
          condition={expression.condition}
          thenExpr={expression.thenExpression}
          elseExpr={expression.elseExpression}
          compact={compact}
        />
      );

    case 'raw':
      return <RawCodeRenderer code={expression.code} />;

    default:
      return <UnknownExpressionRenderer expression={expression} />;
  }
};
```

### 3.2 各子表达式渲染器示例

```typescript
// 字面量渲染器
const LiteralRenderer: React.FC<{ value: any }> = ({ value }) => {
  const getDisplayValue = () => {
    if (value === null) return <span className="text-gray-400">null</span>;
    if (typeof value === 'string') return <span className="text-green-600">"{value}"</span>;
    if (typeof value === 'number') return <span className="text-blue-600">{value}</span>;
    if (typeof value === 'boolean') return <span className="text-purple-600">{String(value)}</span>;
    return <span>{JSON.stringify(value)}</span>;
  };
  return <span className="literal">{getDisplayValue()}</span>;
};

// 变量渲染器
const VariableRenderer: React.FC<{
  name: string;
  highlighted?: boolean;
  onClick?: () => void;
}> = ({ name, highlighted, onClick }) => (
  <span
    className={`variable cursor-pointer hover:underline ${
      highlighted ? 'bg-yellow-200 font-bold' : 'text-orange-600'
    }`}
    onClick={onClick}
  >
    {name}
  </span>
);

// 方法调用渲染器
const MethodCallRenderer: React.FC<{
  target: Expression;
  methodName: string;
  arguments: Expression[];
  depth: number;
}> = ({ target, methodName, arguments: args, depth }) => (
  <span className="method-call">
    <ExpressionRenderer expression={target} depth={depth + 1} />
    <span className="text-gray-500">.</span>
    <span className="text-blue-700 font-medium">{methodName}</span>
    <span className="text-gray-500">(</span>
    {args.map((arg, i) => (
      <React.Fragment key={i}>
        {i > 0 && <span className="text-gray-500">, </span>}
        <ExpressionRenderer expression={arg} depth={depth + 1} />
      </React.Fragment>
    ))}
    <span className="text-gray-500">)</span>
  </span>
);

// 运算符渲染器
const OperatorRenderer: React.FC<{
  operator: string;
  left: Expression;
  right: Expression | null;
  depth: number;
  compact: boolean;
}> = ({ operator, left, right, depth, compact }) => {
  // 一元运算符
  if (right === null) {
    const isPrefix = ['!', '-', '++_prefix', '--_prefix'].includes(operator);
    const op = operator.replace('_prefix', '').replace('_postfix', '');
    return (
      <span className="unary-operator">
        {isPrefix && <span className="text-red-600">{op}</span>}
        <ExpressionRenderer expression={left} depth={depth + 1} compact={compact} />
        {!isPrefix && <span className="text-red-600">{op}</span>}
      </span>
    );
  }

  // 二元运算符
  return (
    <span className="binary-operator">
      <ExpressionRenderer expression={left} depth={depth + 1} compact={compact} />
      <span className="text-red-600 mx-1">{operator}</span>
      <ExpressionRenderer expression={right} depth={depth + 1} compact={compact} />
    </span>
  );
};
```

## 5. 完整流程展示组件

### 5.1 主画布组件 (使用 React Flow)

```typescript
// FlowCanvas.tsx
import ReactFlow, {
  Node, Edge, Background, Controls, MiniMap
} from 'reactflow';

const FlowCanvas: React.FC<{
  flow: VisualFlowSchema;
  onNodeSelect?: (nodeId: string) => void;
  selectedNodeId?: string;
}> = ({ flow, onNodeSelect, selectedNodeId }) => {

  // 将 VisualNode 转换为 React Flow Node
  const nodes: Node[] = useMemo(() =>
    flow.nodes.map(node => ({
      id: node.id,
      type: getNodeComponentType(node.type), // 映射到自定义组件
      position: node.position || { x: 0, y: 0 },
      data: {
        ...node.data,
        originalNode: node,
        selected: node.id === selectedNodeId
      },
    })),
    [flow.nodes, selectedNodeId]
  );

  // 将 VisualEdge 转换为 React Flow Edge
  const edges: Edge[] = useMemo(() =>
    flow.edges.map(edge => ({
      id: edge.id,
      source: edge.source,
      target: edge.target,
      sourceHandle: edge.sourceHandle,
      targetHandle: edge.targetHandle,
      label: edge.label,
      animated: edge.animated,
      type: getEdgeType(edge.type),
      style: getEdgeStyle(edge),
    })),
    [flow.edges]
  );

  // 自定义节点类型映射
  const nodeTypes = useMemo(() => ({
    startNode: StartEndNode,
    endNode: StartEndNode,
    conditionNode: ConditionNode,
    assignmentNode: AssignmentNode,
    functionCallNode: FunctionCallNode,
    forLoopNode: ForLoopNode,
    foreachLoopNode: ForeachLoopNode,
    whileLoopNode: WhileLoopNode,
    returnNode: ReturnNode,
    expressionNode: ExpressionNode,
    tryCatchNode: TryCatchNode,
    throwNode: ThrowNode,
    breakNode: ControlFlowNode,
    continueNode: ControlFlowNode,
  }), []);

  return (
    <div className="h-full w-full">
      <ReactFlow
        nodes={nodes}
        edges={edges}
        nodeTypes={nodeTypes}
        onNodeClick={(_, node) => onNodeSelect?.(node.id)}
        fitView
      >
        <Background />
        <Controls />
        <MiniMap />
      </ReactFlow>
    </div>
  );
};
```

```typescript
// utils/nodeMapping.ts
const getNodeComponentType = (type: string): string => {
  const mapping: Record<string, string> = {
    'start': 'startNode',
    'end': 'endNode',
    'if': 'conditionNode',
    'assignment': 'assignmentNode',
    'function_call': 'functionCallNode',
    'for': 'forLoopNode',
    'foreach': 'foreachLoopNode',
    'while': 'whileLoopNode',
    'return': 'returnNode',
    'expression': 'expressionNode',
    'try_catch': 'tryCatchNode',
    'throw': 'throwNode',
    'break': 'breakNode',
    'continue': 'continueNode',
  };
  return mapping[type] || 'expressionNode';
};
```

## 7. 组件复用策略总结

### 7.1 高复用组件

| 组件 | 复用场景 | 复用度 |
|------|----------|--------|
| `ExpressionRenderer` | 所有节点、属性面板、代码预览 | ★★★★★ |
| `LiteralRenderer` | 字面量展示、默认值、参数 | ★★★★★ |
| `VariableRenderer` | 变量引用、赋值目标、参数 | ★★★★★ |
| `MethodCallRenderer` | 方法调用、静态方法、链式调用 | ★★★★☆ |
| `BaseNode` | 所有节点类型 | ★★★★☆ |
| `FlowCanvas` | 主流程、函数体、子流程 | ★★★★☆ |

### 7.2 复用设计原则

1. **表达式递归渲染**: 所有复杂表达式都通过 `ExpressionRenderer` 递归展开
2. **节点类型映射**: 通过 `nodeTypes` 配置动态映射节点组件
3. **样式抽象**: 通过 `BaseNode` 统一节点样式，子组件只关注内容
4. **数据驱动**: 组件只依赖传入的数据结构，不关心数据来源
5. **可组合性**: 小组件可自由组合成更复杂的展示

```typescript
// api/parseService.ts
export const parseToVisual = async (script: string): Promise<ParseToVisualResponse> => {
  const response = await fetch('/api/v1/parse/to-visual', {
    method: 'POST',
    headers: { 'Content-Type': 'application/json' },
    body: JSON.stringify({ script }),

  return response.json();

// 使用示例
const App = () => {
  const [flow, setFlow] = useState<VisualFlowSchema | null>(null);

  const handleParse = async (script: string) => {
    const result = await parseToVisual(script);
    if (result.success) {
      setFlow(result.data.flow);
    }
  };

  return flow ? <QLExpressVisualEditor flow={flow} /> : <ScriptInput onParse={handleParse} />;
};
```

### 10.3 Schema 验证器

```typescript
// utils/schemaValidator.ts

interface ValidationResult {
  valid: boolean;
  errors: string[];
  warnings: string[];
}

// 节点类型白名单
const VALID_NODE_TYPES = [
  'start', 'end', 'if', 'for', 'foreach', 'while',
  'assignment', 'function_call', 'expression', 'return',
  'break', 'continue', 'try_catch', 'throw'
];

// 表达式类型白名单
const VALID_EXPRESSION_TYPES = [
  'literal', 'variable', 'operator', 'function', 'method',
  'static_method', 'new', 'cast', 'field_access', 'array_access',
  'ternary', 'raw'
];

export function validateSchema(data: any): ValidationResult {
  const errors: string[] = [];
  const warnings: string[] = [];

  // 1. 基础结构验证
  if (!data || typeof data !== 'object') {
    errors.push('无效的 JSON 结构');
    return { valid: false, errors, warnings };
  }

  // 2. 版本验证
  if (!data.version) {
    warnings.push('缺少 version 字段，将使用默认版本');
  }

  // 3. 节点验证
  if (!Array.isArray(data.nodes)) {
    errors.push('nodes 必须是数组');
  } else {
    data.nodes.forEach((node: any, index: number) => {
      const nodeErrors = validateNode(node, index);
      errors.push(...nodeErrors);
    });

    // 验证必须有 start 和 end 节点
    const hasStart = data.nodes.some((n: any) => n.type === 'start');
    const hasEnd = data.nodes.some((n: any) => n.type === 'end');
    if (!hasStart) warnings.push('建议添加 start 节点');
    if (!hasEnd) warnings.push('建议添加 end 节点');
  }

  // 4. 边验证
  if (!Array.isArray(data.edges)) {
    errors.push('edges 必须是数组');
  } else {
    const nodeIds = new Set(data.nodes?.map((n: any) => n.id) || []);
    data.edges.forEach((edge: any, index: number) => {
      const edgeErrors = validateEdge(edge, index, nodeIds);
      errors.push(...edgeErrors);
    });
  }

  // 5. 函数验证
  if (data.functions && Array.isArray(data.functions)) {
    data.functions.forEach((func: any, index: number) => {
      const funcErrors = validateFunction(func, index);
      errors.push(...funcErrors);
    });
  }

  // 6. 导入验证
  if (data.imports && !Array.isArray(data.imports)) {
    errors.push('imports 必须是字符串数组');
  }

  return {
    valid: errors.length === 0,
    errors,
    warnings
  };
}

function validateNode(node: any, index: number): string[] {
  const errors: string[] = [];
  const prefix = `节点[${index}]`;

  if (!node.id) {
    errors.push(`${prefix}: 缺少 id`);
  }

  if (!node.type) {
    errors.push(`${prefix}: 缺少 type`);
  } else if (!VALID_NODE_TYPES.includes(node.type)) {
    errors.push(`${prefix}: 无效的节点类型 "${node.type}"`);
  }

  // 验证特定节点类型的 data 结构
  if (node.data) {
    const dataErrors = validateNodeData(node.type, node.data, prefix);
    errors.push(...dataErrors);
  }

  return errors;
}

function validateNodeData(type: string, data: any, prefix: string): string[] {
  const errors: string[] = [];

  switch (type) {
    case 'if':
      if (!data.condition) {
        errors.push(`${prefix}: if 节点缺少 condition`);
      } else {
        errors.push(...validateExpression(data.condition, `${prefix}.condition`));
      }
      break;

    case 'assignment':
      if (!data.variable) {
        errors.push(`${prefix}: assignment 节点缺少 variable`);
      }
      if (!data.value) {
        errors.push(`${prefix}: assignment 节点缺少 value`);
      } else {
        errors.push(...validateExpression(data.value, `${prefix}.value`));
      }
      break;

    case 'for':
      if (!data.condition) {
        errors.push(`${prefix}: for 节点缺少 condition`);
      }
      break;

    case 'function_call':
      if (!data.expression && !data.functionName) {
        errors.push(`${prefix}: function_call 节点缺少 expression 或 functionName`);
      }
      break;

    case 'return':
      // return 可以没有值
      if (data.value) {
        errors.push(...validateExpression(data.value, `${prefix}.value`));
      }
      break;
  }

  return errors;
}

function validateExpression(expr: any, path: string): string[] {
  const errors: string[] = [];

  if (!expr || typeof expr !== 'object') {
    errors.push(`${path}: 无效的表达式`);
    return errors;
  }

  if (!expr.type) {
    errors.push(`${path}: 表达式缺少 type`);
    return errors;
  }

  if (!VALID_EXPRESSION_TYPES.includes(expr.type)) {
    errors.push(`${path}: 无效的表达式类型 "${expr.type}"`);
  }

  // 递归验证嵌套表达式
  switch (expr.type) {
    case 'operator':
      if (expr.left) {
        errors.push(...validateExpression(expr.left, `${path}.left`));
      }
      if (expr.right) {
        errors.push(...validateExpression(expr.right, `${path}.right`));
      }
      break;

    case 'method':
      if (expr.target) {
        errors.push(...validateExpression(expr.target, `${path}.target`));
      }
      if (expr.arguments) {
        expr.arguments.forEach((arg: any, i: number) => {
          errors.push(...validateExpression(arg, `${path}.arguments[${i}]`));
        });
      }
      break;

    case 'cast':
      if (expr.expression) {
        errors.push(...validateExpression(expr.expression, `${path}.expression`));
      }
      break;

    case 'ternary':
      if (expr.condition) {
        errors.push(...validateExpression(expr.condition, `${path}.condition`));
      }
      if (expr.thenExpression) {
        errors.push(...validateExpression(expr.thenExpression, `${path}.thenExpression`));
      }
      if (expr.elseExpression) {
        errors.push(...validateExpression(expr.elseExpression, `${path}.elseExpression`));
      }
      break;
  }

  return errors;
}

function validateEdge(edge: any, index: number, nodeIds: Set<string>): string[] {
  const errors: string[] = [];
  const prefix = `连线[${index}]`;

  if (!edge.id) {
    errors.push(`${prefix}: 缺少 id`);
  }
  if (!edge.source) {
    errors.push(`${prefix}: 缺少 source`);
  } else if (!nodeIds.has(edge.source)) {
    errors.push(`${prefix}: source 引用了不存在的节点 "${edge.source}"`);
  }
  if (!edge.target) {
    errors.push(`${prefix}: 缺少 target`);
  } else if (!nodeIds.has(edge.target)) {
    errors.push(`${prefix}: target 引用了不存在的节点 "${edge.target}"`);
  }

  return errors;
}

function validateFunction(func: any, index: number): string[] {
  const errors: string[] = [];
  const prefix = `函数[${index}]`;

  if (!func.name) {
    errors.push(`${prefix}: 缺少 name`);
  }

  if (func.body) {
    if (func.body.nodes) {
      func.body.nodes.forEach((node: any, i: number) => {
        const nodeErrors = validateNode(node, i);
        errors.push(...nodeErrors.map(e => `${prefix}.body.${e}`));
      });
    }
  }

  return errors;
}
```

```typescript
// utils/formatConverter.ts

/**
 * 将 VisualFlowSchema 转换为 React Flow 格式
 */
export function schemaToReactFlow(schema: VisualFlowSchema): {
  nodes: Node[];
  edges: Edge[];
} {
  const nodes: Node[] = schema.nodes.map(node => ({
    id: node.id,
    type: mapNodeType(node.type),
    position: node.position || calculateAutoPosition(node, schema),
    data: {
      ...node.data,
      _originalType: node.type,  // 保留原始类型用于导出
      _style: node.style,
    },
    selected: node.selected,
    dragging: node.dragging,
  }));

  const edges: Edge[] = schema.edges.map(edge => ({
    id: edge.id,
    source: edge.source,
    target: edge.target,
    sourceHandle: edge.sourceHandle,
    targetHandle: edge.targetHandle,
    label: edge.label,
    type: mapEdgeType(edge.type),
    animated: edge.animated,
    data: {
      ...edge.data,
      _originalType: edge.type,
    },
    style: edge.style ? {
      stroke: edge.style.color,
      strokeWidth: edge.style.width,
    } : undefined,
  }));

  return { nodes, edges };
}

/**
 * 将 React Flow 状态转换回 VisualFlowSchema
 */
export function reactFlowToSchema(
  originalSchema: VisualFlowSchema,
  nodes: Node[],
  edges: Edge[]
): VisualFlowSchema {
  // 更新节点
  const updatedNodes: VisualNode[] = nodes.map(node => ({
    id: node.id,
    type: node.data._originalType || reverseMapNodeType(node.type),
    position: {
      x: node.position.x,
      y: node.position.y,
    },
    data: cleanNodeData(node.data),
    style: node.data._style,
    selected: node.selected,
    dragging: node.dragging,
  }));

  // 更新边
  const updatedEdges: VisualEdge[] = edges.map(edge => ({
    id: edge.id,
    source: edge.source,
    target: edge.target,
    sourceHandle: edge.sourceHandle || undefined,
    targetHandle: edge.targetHandle || undefined,
    type: edge.data?._originalType || reverseMapEdgeType(edge.type),
    label: typeof edge.label === 'string' ? edge.label : undefined,
    data: edge.data ? cleanEdgeData(edge.data) : undefined,
    style: edge.style ? {
      color: edge.style.stroke,
      width: edge.style.strokeWidth,
    } : undefined,
    animated: edge.animated,
  }));

  return {
    ...originalSchema,
    nodes: updatedNodes,
    edges: updatedEdges,
    metadata: {
      ...originalSchema.metadata,
      updatedAt: new Date().toISOString(),
    },
  };
}

/**
 * 清理节点数据，移除前端特有字段
 */
function cleanNodeData(data: any): NodeData {
  const { _originalType, _style, ...cleanData } = data;
  return cleanData;
}

/**
 * 清理边数据
 */
function cleanEdgeData(data: any): EdgeData | undefined {
  if (!data) return undefined;
  const { _originalType, ...cleanData } = data;
  return Object.keys(cleanData).length > 0 ? cleanData : undefined;
}

/**
 * 节点类型映射: Schema -> React Flow
 */
function mapNodeType(type: string): string {
  const mapping: Record<string, string> = {
    'start': 'startNode',
    'end': 'endNode',
    'if': 'conditionNode',
    'for': 'forLoopNode',
    'foreach': 'foreachLoopNode',
    'while': 'whileLoopNode',
    'assignment': 'assignmentNode',
    'function_call': 'functionCallNode',
    'expression': 'expressionNode',
    'return': 'returnNode',
    'break': 'breakNode',
    'continue': 'continueNode',
    'try_catch': 'tryCatchNode',
    'throw': 'throwNode',
  };
  return mapping[type] || 'expressionNode';
}

/**
 * 节点类型反向映射: React Flow -> Schema
 */
function reverseMapNodeType(type: string): string {
  const mapping: Record<string, string> = {
    'startNode': 'start',
    'endNode': 'end',
    'conditionNode': 'if',
    'forLoopNode': 'for',
    'foreachLoopNode': 'foreach',
    'whileLoopNode': 'while',
    'assignmentNode': 'assignment',
    'functionCallNode': 'function_call',
    'expressionNode': 'expression',
    'returnNode': 'return',
    'breakNode': 'break',
    'continueNode': 'continue',
    'tryCatchNode': 'try_catch',
    'throwNode': 'throw',
  };
  return mapping[type] || 'expression';
}

/**
 * 自动计算节点位置 (当节点没有位置信息时)
 */
function calculateAutoPosition(
  node: VisualNode,
  schema: VisualFlowSchema
): Position {
  // 简单的自动布局算法
  const index = schema.nodes.indexOf(node);
  const col = index % 3;
  const row = Math.floor(index / 3);

  return {
    x: 100 + col * 300,
    y: 50 + row * 150,
  };
}
```

## 11. 完整数据流总结

| 操作 | 输入 | 处理流程 | 输出 |
|------|------|----------|------|
| **导入JSON** | `.json`文件 | 解析 → 验证 → 转换为ReactFlow格式 → 更新Store | 画布展示 |
| **导入QL** | QL脚本文本 | 调用后端API → 获取JSON → 同上 | 画布展示 |
| **导出JSON** | Store中的nodes/edges | 收集状态 → 转换为Schema格式 → 序列化 | `.json`文件下载 |
| **导出QL** | Store中的nodes/edges | 收集状态 → 转换为Schema → 调用后端API | `.ql`文件下载 |
| **实时编辑** | 用户拖拽/修改 | ReactFlow事件 → 同步到Store | isDirty标记 |
