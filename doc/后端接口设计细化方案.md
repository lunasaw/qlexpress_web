
#### 2.1.2 QL脚本转可视化JSON

```java

 *
 * 将QLExpress脚本解析为可视化流程JSON
 */
@PostMapping("/to-json")
public ResponseEntity<TranspileToJsonResponse> transpileToJson(
    @Valid @RequestBody TranspileToJsonRequest request) {
    // ...
}
```

**请求结构：**

```java
@Data
public class TranspileToJsonRequest {

    @NotBlank(message = "脚本不能为空")
    @Size(max = 100000, message = "脚本长度不能超过100KB")
    private String script;

    /**
     * 布局选项
     */
    private LayoutOptions layout;
}

@Data
public class LayoutOptions {
    /**
     * 布局算法
     */
    private LayoutAlgorithm algorithm = LayoutAlgorithm.DAGRE;

    /**
     * 节点间距
     */
    private int nodeSpacing = 100;

    /**
     * 层级间距
     */
    private int rankSpacing = 80;

    /**
     * 布局方向
     */
    private LayoutDirection direction = LayoutDirection.TB;
}

public enum LayoutAlgorithm {
    DAGRE,      // 有向图布局
    TREE,       // 树形布局
    FORCE,      // 力导向布局
    MANUAL      // 手动布局（保持原位置）
}

public enum LayoutDirection {
    TB,  // Top to Bottom
    BT,  // Bottom to Top
    LR,  // Left to Right
    RL   // Right to Left
}
```

**响应结构：**

```java
@Data
public class TranspileToJsonResponse {
    /**
     * 可视化流程定义
     */
    private VisualFlowSchema flow;

    /**
     * 解析警告
     */
    private List<ParseWarning> warnings;

    /**
     * 不支持的语法元素（无法可视化的部分）
     */
    private List<UnsupportedElement> unsupportedElements;
}

@Data
public class UnsupportedElement {
    private String type;
    private String description;
    private int line;
    private int column;
    private String rawCode;
}
```

```java
@Service
@Slf4j
public class TranspileService {

    private final VisualToQLTranspiler visualToQLTranspiler;
    private final QLToVisualParser qlToVisualParser;
    private final FlowValidator flowValidator;
    private final Express4Runner express4Runner;

    /**
     * JSON → QL脚本
     */
    public TranspileToQLResponse transpileToQL(VisualFlowSchema flow,
                                                TranspileOptions options) {
        // 1. 验证流程结构
        ValidateFlowResponse validation = flowValidator.validate(flow);
        if (!validation.isValid()) {
            throw new TranspileException("流程验证失败", validation.getErrors());
        }

        // 2. 转译
        TranspileContext context = new TranspileContext(options);
        String script = visualToQLTranspiler.transpile(flow, context);

        // 3. 语法验证生成的脚本
        try {
            express4Runner.parseToSyntaxTree(script);
        } catch (Exception e) {
            throw new TranspileException("生成的脚本语法错误", e);
        }

        // 4. 构建响应
        return TranspileToQLResponse.builder()
            .script(script)
            .sourceMap(context.getSourceMap())
            .warnings(context.getWarnings())
            .stats(context.getStats())
            .build();
    }

    /**
     * QL脚本 → JSON
     */
    public TranspileToJsonResponse transpileToJson(String script,
                                                    LayoutOptions layout) {
        // 1. 解析脚本为AST
        QLParser.ProgramContext ast;
        try {
            ast = express4Runner.parseToSyntaxTree(script);
        } catch (Exception e) {
            throw new TranspileException("脚本语法错误", e);
        }

        // 2. 转换为可视化模型
        ParseContext context = new ParseContext(layout);
        VisualFlowSchema flow = qlToVisualParser.parse(ast, context);

        // 3. 应用布局
        applyLayout(flow, layout);

        // 4. 构建响应
        return TranspileToJsonResponse.builder()
            .flow(flow)
            .warnings(context.getWarnings())
            .unsupportedElements(context.getUnsupportedElements())
            .build();
    }
```

```java
@Component
public class VisualToQLTranspiler {

    private final NodeRegistry nodeRegistry;

    public String transpile(VisualFlowSchema schema, TranspileContext context) {
        StringBuilder sb = new StringBuilder();

        // 1. 生成import
        generateImports(schema.getImports(), sb, context);

        // 2. 生成全局变量
        generateVariables(schema.getVariables(), sb, context);

        // 3. 生成自定义函数
        generateFunctions(schema.getFunctions(), sb, context);

        // 4. 构建图结构
        FlowGraph graph = buildGraph(schema.getNodes(), schema.getEdges());

        // 5. 拓扑排序并检测循环
        List<VisualNode> sortedNodes = topologicalSort(graph);

        // 6. 生成主逻辑
        generateMainFlow(sortedNodes, graph, sb, context);

        return context.isFormat() ? format(sb.toString()) : sb.toString();
    }

    private void generateMainFlow(List<VisualNode> nodes, FlowGraph graph,
                                   StringBuilder sb, TranspileContext context) {
        // 找到起始节点
        VisualNode startNode = findStartNode(nodes);

        // 使用深度优先遍历生成代码
        Set<String> visited = new HashSet<>();
        generateNode(startNode, graph, sb, context, visited, 0);
    }

    private void generateNode(VisualNode node, FlowGraph graph,
                              StringBuilder sb, TranspileContext context,
                              Set<String> visited, int depth) {
        if (visited.contains(node.getId())) {
            return;
        }
        visited.add(node.getId());

        // 记录源映射
        int startLine = context.getCurrentLine();

        // 获取节点处理器
        NodeTranspiler transpiler = nodeRegistry.getTranspiler(node.getType());
        if (transpiler == null) {
            context.addWarning(node.getId(), "UNKNOWN_NODE_TYPE",
                "未知的节点类型: " + node.getType());
            return;
        }

        // 生成节点代码
        NodeTranspileResult result = transpiler.transpile(node, context);

        // 添加到输出
        appendWithIndent(sb, result.getCode(), depth, context);

        // 记录源映射
        int endLine = context.getCurrentLine();
        context.addSourceMapping(node.getId(), startLine, endLine);

        // 处理后续节点
        if (result.hasCustomFlow()) {
            // 节点自行控制流程（如if、for）
            result.generateChildren(graph, sb, context, visited, depth);
        } else {
            // 标准顺序执行
            for (VisualEdge edge : graph.getOutEdges(node.getId())) {
                VisualNode nextNode = graph.getNode(edge.getTarget());
                generateNode(nextNode, graph, sb, context, visited, depth);
            }
        }
    }
}
```

```java
    @Override
    public NodeTranspileResult transpile(VisualNode node, TranspileContext context) {

        StringBuilder code = new StringBuilder();
        code.append("if (").append(expressionToQL(data.getCondition())).append(") {");

        return NodeTranspileResult.builder()

            .hasCustomFlow(true)
            .childGenerator((graph, sb, ctx, visited, depth) -> {

                    generateBranch(data.getThenBranch(), graph, sb, ctx, visited, depth + 1);
                    ctx.dedent();

                    appendLine(sb, "} else {", depth, ctx);
                    ctx.indent();
                    generateBranch(data.getElseBranch(), graph, sb, ctx, visited, depth + 1);
                    ctx.dedent();
                }

                appendLine(sb, "}", depth, ctx);
            })
            .build();
    }
```

```java
        List<VisualNode> nodes = new ArrayList<>();

        int y = 150;

        if (ast.blockStatements() != null) {
            for (QLParser.BlockStatementContext stmt : ast.blockStatements().blockStatement()) {
                ParseResult result = parseStatement(stmt, nodes, edges, 100, y, context);

            }
        }

        // 添加结束节点
        VisualNode endNode = createNode("end", "结束", 100, y);
        nodes.add(endNode);
        edges.add(createEdge(prevNodeId, endNode.getId()));

        schema.setNodes(nodes);
        schema.setEdges(edges);

        return schema;
    }

    private ParseResult parseStatement(QLParser.BlockStatementContext stmt,

        // 判断语句类型并解析
        if (isIfStatement(stmt)) {
            return parseIfStatement(stmt, nodes, edges, x, y, context);
        } else if (isForStatement(stmt)) {
            return parseForStatement(stmt, nodes, edges, x, y, context);
        } else if (isWhileStatement(stmt)) {
            return parseWhileStatement(stmt, nodes, edges, x, y, context);
        } else if (isReturnStatement(stmt)) {
            return parseReturnStatement(stmt, nodes, x, y, context);
        } else if (isExpressionStatement(stmt)) {
            return parseExpressionStatement(stmt, nodes, x, y, context);

            context.addUnsupportedElement(stmt.getText(), stmt.getStart().getLine());

                parseExpression(getTarget(expr)),
                getMethodName(expr),
                getArguments(expr).stream()
                    .map(this::parseExpression)
                    .collect(Collectors.toList())
            );
        }

        // 无法解析，使用raw
        return Expression.raw(expr.getText());
```

```java
@Service
@Slf4j
public class ExecuteService {

    private final Express4Runner express4Runner;
    private final TranspileService transpileService;
    private final TraceService traceService;

    /**
     * 执行脚本
     */
    public ExecuteResponse executeScript(String script,
                                          Map<String, Object> context,
                                          ExecuteOptions options) {
        long startTime = System.currentTimeMillis();

        // 构建执行选项
        QLOptions qlOptions = buildQLOptions(options);

        // 创建执行上下文
        ExpressContext expressContext = new DefaultExpressContext(context);

        try {
            // 执行脚本
            QLResult result;
            if (options.isEnableTrace()) {
                result = executeWithTrace(script, expressContext, qlOptions);
            } else {
                result = express4Runner.execute(script, expressContext, qlOptions);
            }

            // 构建响应
            return ExecuteResponse.builder()
                .success(true)
                .result(result.getResult())
                .resultType(getTypeName(result.getResult()))
                .traces(options.isEnableTrace() ? traceService.getTraces() : null)
                .stats(buildStats(startTime, options))
                .outputContext(extractOutputContext(expressContext))
                .build();

        } catch (QLException e) {
            log.error("脚本执行失败: {}", e.getMessage(), e);
            return ExecuteResponse.builder()
                .success(false)
                .error(buildError(e))
                .stats(buildStats(startTime, options))
                .build();
        } catch (Exception e) {
            log.error("脚本执行异常: {}", e.getMessage(), e);
            return ExecuteResponse.builder()
                .success(false)
                .error(ExecuteError.builder()
                    .type(e.getClass().getSimpleName())
                    .message(e.getMessage())
                    .build())
                .stats(buildStats(startTime, options))
                .build();
        }
    }

    /**
     * 执行可视化流程
     */
    public ExecuteFlowResponse executeFlow(VisualFlowSchema flow,
                                            Map<String, Object> context,
                                            ExecuteOptions options,
                                            boolean includeNodeDetails) {
        // 1. 先转译为QL脚本
        TranspileToQLResponse transpileResult = transpileService.transpileToQL(
            flow, new TranspileOptions());

        // 2. 执行脚本
        ExecuteResponse executeResult = executeScript(
            transpileResult.getScript(), context, options);

        // 3. 构建响应
        ExecuteFlowResponse response = new ExecuteFlowResponse();
        BeanUtils.copyProperties(executeResult, response);

        if (includeNodeDetails && options.isEnableTrace()) {
            response.setNodeDetails(
                traceService.mapTracesToNodes(flow, transpileResult.getSourceMap())
            );
            response.setExecutionPath(
                traceService.extractExecutionPath(flow)
            );
        }

        return response;
    }

    private QLOptions buildQLOptions(ExecuteOptions options) {
        return QLOptions.builder()
            .timeoutMillis(options.getTimeout())
            .cache(options.isEnableCache())
            .maxLoopCount(options.getMaxLoopCount())
            .preciseBigDecimal(options.isPreciseBigDecimal())
            .traceExpression(options.isEnableTrace())
            .shortCircuit(options.isShortCircuit())
            .build();
    }

    private QLResult executeWithTrace(String script,
                                       ExpressContext context,
                                       QLOptions qlOptions) {
        // 启用追踪模式执行
        traceService.startTrace();
        try {
            return express4Runner.execute(script, context, qlOptions);
        } finally {
            traceService.endTrace();
        }
    }
}
```

```java
@Service
public class ValidateService {

    private final Express4Runner express4Runner;

    /**
     * 验证脚本
     */
    public ValidateScriptResponse validateScript(String script,
                                                   boolean checkSemantics,
                                                   boolean checkBestPractices) {
        List<SyntaxError> syntaxErrors = new ArrayList<>();
        List<SemanticWarning> semanticWarnings = new ArrayList<>();
        List<BestPracticeSuggestion> suggestions = new ArrayList<>();

        // 1. 语法检查
        try {
            express4Runner.parseToSyntaxTree(script);
        } catch (QLSyntaxException e) {
            syntaxErrors.add(SyntaxError.builder()
                .line(e.getLine())
                .column(e.getColumn())
                .message(e.getMessage())
                .code("SYNTAX_ERROR")
                .build());
        }

        if (!syntaxErrors.isEmpty()) {
            return ValidateScriptResponse.builder()
                .valid(false)
                .syntaxErrors(syntaxErrors)
                .build();
        }

        // 2. 语义检查
        if (checkSemantics) {
            semanticWarnings.addAll(checkSemantics(script));
        }

        // 3. 最佳实践检查
        if (checkBestPractices) {
            suggestions.addAll(checkBestPractices(script));
        }

        // 4. 复杂度分析
        ComplexityAnalysis complexity = analyzeComplexity(script);

        return ValidateScriptResponse.builder()
            .valid(true)
            .syntaxErrors(syntaxErrors)
            .semanticWarnings(semanticWarnings)
            .suggestions(suggestions)
            .complexity(complexity)
            .build();
    }

    /**
     * 验证流程
     */
    public ValidateFlowResponse validateFlow(VisualFlowSchema flow,
                                              ValidationLevel level) {
        List<FlowError> structureErrors = new ArrayList<>();
        List<NodeError> nodeErrors = new ArrayList<>();
        List<EdgeError> edgeErrors = new ArrayList<>();
        List<TypeError> typeErrors = new ArrayList<>();

        // 1. 结构验证
        structureErrors.addAll(validateStructure(flow));

        // 2. 节点验证
        for (VisualNode node : flow.getNodes()) {
            nodeErrors.addAll(validateNode(node));
        }

        // 3. 连线验证
        for (VisualEdge edge : flow.getEdges()) {
            edgeErrors.addAll(validateEdge(edge, flow));
        }

        // 4. 类型验证（严格模式）
        if (level == ValidationLevel.STRICT) {
            typeErrors.addAll(typeChecker.checkFlow(flow));
        }

        boolean valid = structureErrors.isEmpty()
            && nodeErrors.isEmpty()
            && edgeErrors.isEmpty()
            && typeErrors.isEmpty();

        return ValidateFlowResponse.builder()
            .valid(valid)
            .structureErrors(structureErrors)
            .nodeErrors(nodeErrors)
            .edgeErrors(edgeErrors)
            .typeErrors(typeErrors)
            .build();
    }

    /**
     * 验证连接
     */
    public ConnectionValidateResponse validateConnection(
            VisualNode sourceNode, String sourceHandle,
            VisualNode targetNode, String targetHandle) {

        TypeCompatibility compatibility = typeChecker.checkConnection(
            sourceNode, sourceHandle, targetNode, targetHandle);

        return ConnectionValidateResponse.builder()
            .valid(compatibility.isCompatible())
            .message(compatibility.isCompatible() ? "连接有效" :
                "类型不兼容: " + compatibility.getSourceType() +
                " → " + compatibility.getTargetType())
            .typeCompatibility(compatibility)
            .requiresConversion(
                compatibility.getLevel() == CompatibilityLevel.EXPLICIT)
            .conversionSuggestion(
                compatibility.getLevel() == CompatibilityLevel.EXPLICIT ?
                    "使用 (" + compatibility.getTargetType() + ") 进行类型转换" : null)
            .build();
    }

    private List<FlowError> validateStructure(VisualFlowSchema flow) {
        List<FlowError> errors = new ArrayList<>();

        // 检查起始节点
        long startCount = flow.getNodes().stream()
            .filter(n -> "start".equals(n.getType()))
            .count();
        if (startCount == 0) {
            errors.add(new FlowError("NO_START_NODE", "缺少起始节点",
                FlowErrorType.NO_START_NODE));
        } else if (startCount > 1) {
            errors.add(new FlowError("MULTIPLE_START_NODES", "存在多个起始节点",
                FlowErrorType.MULTIPLE_START_NODES));
        }

        // 检查不可达节点
        Set<String> reachable = findReachableNodes(flow);
        for (VisualNode node : flow.getNodes()) {
            if (!reachable.contains(node.getId()) && !"start".equals(node.getType())) {
                errors.add(new FlowError("UNREACHABLE_NODE",
                    "节点 " + node.getId() + " 不可达",
                    FlowErrorType.UNREACHABLE_NODES));
            }
        }

        // 检查无限循环
        if (hasInfiniteLoop(flow)) {
            errors.add(new FlowError("INFINITE_LOOP", "检测到可能的无限循环",
                FlowErrorType.INFINITE_LOOP));
        }

        return errors;
    }
}
```

```java
        List<SecurityIssue> issues = new ArrayList<>();

        // 检查危险函数调用
        checkDangerousCalls(script, issues);

        // 检查类型导入
        checkImports(script, issues);

        // 检查反射使用
        checkReflection(script, issues);

        return SecurityCheckResult.builder()
            .safe(issues.isEmpty())
            .issues(issues)
            .build();
    }

    private void checkDangerousCalls(String script, List<SecurityIssue> issues) {
        // 检查Runtime.exec
        if (script.contains("Runtime") || script.contains("ProcessBuilder")) {
            issues.add(new SecurityIssue("DANGEROUS_CALL",
                "禁止使用Runtime/ProcessBuilder", SecurityLevel.CRITICAL));
        }

        // 检查文件操作
        if (script.contains("File") || script.contains("FileOutputStream")) {
            issues.add(new SecurityIssue("FILE_ACCESS",
                "禁止直接文件操作", SecurityLevel.HIGH));
        }

        // 检查网络操作
        if (script.contains("URL") || script.contains("Socket")) {
            issues.add(new SecurityIssue("NETWORK_ACCESS",
                "禁止直接网络操作", SecurityLevel.HIGH));
        }
    }

    private void checkImports(String script, List<SecurityIssue> issues) {
        Pattern importPattern = Pattern.compile("import\\s+([\\w.]+);");
        Matcher matcher = importPattern.matcher(script);

        while (matcher.find()) {
            String importClass = matcher.group(1);

            // 检查是否在白名单
            boolean allowed = allowedPackages.stream()
                .anyMatch(pkg -> importClass.startsWith(pkg));

            // 检查是否在黑名单
            boolean blocked = blockedClasses.stream()
                .anyMatch(cls -> importClass.equals(cls) || importClass.startsWith(cls));

            if (blocked || !allowed) {
                issues.add(new SecurityIssue("BLOCKED_IMPORT",
                    "禁止导入: " + importClass, SecurityLevel.HIGH));
            }
        }
    }
}
```
